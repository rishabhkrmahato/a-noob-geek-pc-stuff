//
// NETFLIX -> SIMKL VIEWING HISTORY EXPORTER
//
// Created/tested: 19 October 2026
// Website: Netflix Viewing Activity page (version/layout observed on 19 October 2026).
// Note: Netflix does not expose a simple public website version number for this page.
//
// DISCLAIMER:
// This script relies on Netflix's current page structure and may stop working
// if Netflix changes its website, HTML elements, or button behaviour.
// Successful export does not guarantee that the entire viewing history was captured.
//
// HOW TO USE:
// 1. Open Netflix in Google Chrome and go to your Viewing Activity page.
// 2. Press F12 and select the Console tab.
// 3. Copy and paste the entire script below, then press Enter.
// 4. Keep the tab open while the script runs.
// 5. The CSV file downloads automatically as netflix-ratings-seen.csv.
// 6. Import it into Simkl: https://simkl.com/apps/import/csv/
//
// OUTPUT: One column -> Title. No ratings are exported.
// This script is intended for the Netflix page layout observed on the date above.
//
(async function () {
    console.log("Starting Netflix title extraction...");

    const MAX_LOOPS = 300;
    const WAIT_AFTER_CLICK = 3000;
    const RETRY_WAIT = 2000;
    const MAX_STALLED_LOOPS = 4;

    const TITLE_SELECTOR =
        '.retip-title, .title, .rowTitle, .history-title, a[href*="/title/"]';

    const sleep = ms =>
        new Promise(resolve => setTimeout(resolve, ms));

    const getTitleElements = () =>
        Array.from(document.querySelectorAll(TITLE_SELECTOR));

    const getTitleCount = () => getTitleElements().length;

    let previousCount = getTitleCount();
    let stalledLoops = 0;
    let totalLoops = 0;

    while (totalLoops < MAX_LOOPS) {
        window.scrollTo(0, document.body.scrollHeight);
        await sleep(500);

        // Find a visible element whose exact text is "Show More".
        const showMoreBtn = Array.from(
            document.querySelectorAll('button, div, span, a')
        ).find(el => {
            const text = (el.innerText || '').trim().toLowerCase();
            return el.offsetHeight > 0 && text === 'show more';
        });

        if (showMoreBtn) {
            console.log(
                `Loading more titles... Current elements: ${previousCount}`
            );

            // Trigger the click using the existing approach.
            showMoreBtn.dispatchEvent(
                new MouseEvent('mousedown', {
                    bubbles: true,
                    cancelable: true,
                    view: window
                })
            );

            showMoreBtn.click();

            showMoreBtn.dispatchEvent(
                new MouseEvent('mouseup', {
                    bubbles: true,
                    cancelable: true,
                    view: window
                })
            );

            // Wait for Netflix to load additional content.
            await sleep(WAIT_AFTER_CLICK);
        } else {
            // Allow time for a temporarily missing button or delayed load.
            await sleep(RETRY_WAIT);
        }

        // IMPORTANT: Count again AFTER waiting for the page to update.
        const currentCount = getTitleCount();

        if (currentCount > previousCount) {
            console.log(`Title elements found: ${currentCount}`);
            stalledLoops = 0;
        } else {
            stalledLoops++;
            console.log(
                `No increase in title elements (${stalledLoops}/${MAX_STALLED_LOOPS}).`
            );
        }

        if (stalledLoops >= MAX_STALLED_LOOPS) {
            console.log(
                'No further titles detected. Finishing extraction.'
            );
            break;
        }

        previousCount = currentCount;
        totalLoops++;
    }

    if (totalLoops >= MAX_LOOPS) {
        console.warn('Reached the maximum loop limit.');
    }

    console.log('Preparing CSV...');

    const uniqueTitles = new Set();
    const ignoredTitles = new Set([
        'show more',
        'back to your account'
    ]);

    getTitleElements().forEach(el => {
        const title = (el.innerText || '').trim();

        if (
            title &&
            !ignoredTitles.has(title.toLowerCase())
        ) {
            uniqueTitles.add(title);
        }
    });

    // Escape quotation marks according to CSV rules.
    const escapeCSV = value =>
        '"' + value.replace(/"/g, '""') + '"';

    const output = [
        'Title',
        ...Array.from(uniqueTitles, escapeCSV)
    ].join('\r\n') + '\r\n';

    console.log(`Exported ${uniqueTitles.size} unique titles.`);

    if (uniqueTitles.size === 0) {
        console.error(
            'No titles were found. Check that you are on the correct Netflix page.'
        );
        return;
    }

    const blob = new Blob(
        ['\uFEFF', output],
        { type: 'text/csv;charset=utf-8;' }
    );

    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');

    link.href = url;
    link.download = 'netflix-ratings-seen.csv';
    link.style.display = 'none';

    document.body.appendChild(link);
    link.click();
    link.remove();

    // Revoke the temporary download URL after a short delay.
    setTimeout(() => URL.revokeObjectURL(url), 1000);

    console.log('CSV download initiated successfully!');
})();
